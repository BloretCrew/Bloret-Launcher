    def load_account_data(self):
        """
        从 config.json 读取当前选中的 Minecraft 账户信息。
        替代原有的 load_cmcl_data。
        """
        log(f"开始从 config.json 读取账户数据")
        try:
            config_data = cfg.read()
            mc_account = config_data.get("MinecraftAccount", {})
            accounts = mc_account.get("accounts", [])
            chosen_idx = mc_account.get("chosen", 0)
            
            if not accounts or chosen_idx < 0 or chosen_idx >= len(accounts):
                self.player_name = i18nText("未登录")
                self.login_mod = i18nText("请在下方登录")
                log(i18nText("config.json 中没有有效的账户信息"))
                return

            # 获取当前选中的账户
            account = accounts[chosen_idx]
            
            self.player_name = account.get('username', i18nText('未登录'))
            acc_type = account.get('type', 'Offline')
            
            # 更新登录方式描述
            if acc_type == "Microsoft":
                self.login_mod = i18nText("微软登录")
            else:
                self.login_mod = i18nText("离线登录")

            log(f"读取到的 playerName: {self.player_name}")
            log(f"读取到的 loginType: {self.login_mod}")
            
            return self.player_name
            
        except Exception as e:
            handle_exception(e)
            log(f"读取账户数据失败: {e}", logging.ERROR)
            self.player_name = i18nText("未登录")

    def on_login_finished(self, widget, success, message):
        # 添加有效性检查
        if hasattr(self, 'login_tip') and self.login_tip and not sip.isdeleted(self.login_tip):
            try:
                self.login_tip.close()
            except RuntimeError:
                pass
        
        # 处理结果
        if success:
            # --- 新增逻辑：从 cmcl.json 导入新登录的账户到 config.json ---
            try:
                if os.path.exists('cmcl.json'):
                    with open('cmcl.json', 'r', encoding='utf-8') as f:
                        cmcl_data = json.load(f)
                    
                    if cmcl_data.get('accounts'):
                        new_acc_cmcl = cmcl_data['accounts'][0]
                        
                        # 构建要保存到 config.json 的账户格式
                        new_account = {
                            "username": new_acc_cmcl.get("playerName"),
                            "uuid": new_acc_cmcl.get("uuid"),
                            "type": "Microsoft" if new_acc_cmcl.get("loginMethod") == 2 else "Offline",
                            "login_time": time.strftime("%Y-%m-%dT%H:%M:%S.000Z", time.gmtime()),
                            # 保存 token 以便后续使用
                            "accessToken": new_acc_cmcl.get("accessToken", ""),
                            "refreshToken": new_acc_cmcl.get("refreshToken", "") # 如果 cmcl 返回了 refresh token
                        }
                        
                        # 读取当前配置
                        config_data = cfg.read()
                        if "MinecraftAccount" not in config_data:
                            config_data["MinecraftAccount"] = {"logined": True, "chosen": 0, "accounts": []}
                        
                        mc_config = config_data["MinecraftAccount"]
                        if "accounts" not in mc_config:
                            mc_config["accounts"] = []
                            
                        # 检查是否存在同名同类型账户，存在则更新，不存在则追加
                        found = False
                        for idx, acc in enumerate(mc_config["accounts"]):
                            if acc.get("uuid") == new_account["uuid"]:
                                mc_config["accounts"][idx] = new_account # 更新
                                mc_config["chosen"] = idx # 选中它
                                found = True
                                break
                        
                        if not found:
                            mc_config["accounts"].append(new_account)
                            mc_config["chosen"] = len(mc_config["accounts"]) - 1 # 选中新添加的
                            
                        mc_config["logined"] = True
                        
                        # 保存
                        with open(BLglobals.config_path, 'w', encoding='utf-8') as f:
                            json.dump(config_data, f, ensure_ascii=False, indent=4)
                        
                        log("已将新登录账户导入 config.json")
            except Exception as e:
                log(f"导入登录数据失败: {e}", logging.ERROR)
            
            # 刷新数据和UI
            self.load_account_data()
            self.update_passport_ui(widget)
            
            # 刷新主页头像等
            self.refresh_home_minecraft_account(self.player_name, self.homeInterface)
            
            # 如果能获取到 refresh_minecraft_accounts 最好，否则可能需要重启软件才能在列表看到新账户
            # 这里简单提示
            InfoBar.success(
                title=i18nText('✅ 登录成功'),
                content=i18nText('登录成功，账户已保存'),
                parent=self
            )
        else:
            InfoBar.error(
                title=i18nText('❎ 登录失败'),
                content=message,
                parent=self
            )

    def update_passport_ui(self, widget):
        # 更新UI显示
        login_way_combo = widget.findChild(ComboBox, "player_login_way")
        name_combo = widget.findChild(ComboBox, "playername")
        
        # 使用 self.player_name 和 self.login_mod (由 load_account_data 设置)
        
        # 更新登录方式
        if login_way_combo:
            login_way_combo.clear()
            login_way_combo.addItem(self.login_mod)
        
        # 更新玩家名称
        if name_combo:
            name_combo.clear()
            name_combo.addItem(self.player_name)
